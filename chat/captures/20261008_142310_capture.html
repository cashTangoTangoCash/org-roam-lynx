this complete project consists of the python script as well as this front-end shell script:

#!/usr/bin/env bash

Paths

DB_PATH="${XDG_CONFIG_HOME:-$HOME/.config}/emacs/org-roam.db" [[ ! -f "$DB_PATH" ]] && DB_PATH="$HOME/.emacs.d/org-roam.db" PYTHON_NAV_SCRIPT="${ROAM_LYNX_PY:-$HOME/.local/bin/roam_lynx.py}"

--- HELP HANDLERS ---

show_brief_help() { cat << 'EOF' Usage: roam-lynx-find [OPTION] [SEARCH_TERM]

Search Org-roam nodes/aliases using fzf and open the selection in the terminal Lynx-style node browser.

Options: -h, --help Show this brief command-line help message and exit.

Examples: roam-lynx-find Interactive fzf search through all notes (sorted by mtime). roam-lynx-find bashrc Pre-filtered fzf search for "bashrc" before entering browser. EOF }

case "$1" in -h|--help) show_brief_help; exit 0 ;; esac

Validations

[[ ! -f "$DB_PATH" ]] && { echo "Error: Database not found at $DB_PATH" >&2; exit 1; } [[ ! -f "$PYTHON_NAV_SCRIPT" ]] && { echo "Error: Python navigator not found at $PYTHON_NAV_SCRIPT" >&2; exit 1; }

SEARCH_TERM="$1"

Base SQL incorporating LEFT JOIN on aliases

BASE_SQL="SELECT nodes.title, COALESCE(GROUP_CONCAT(aliases.alias, ' | '), '') AS alias_list, nodes.file FROM nodes JOIN files ON nodes.file = files.file LEFT JOIN aliases ON nodes.id = aliases.node_id WHERE nodes.title IS NOT NULL AND nodes.title != '' GROUP BY nodes.id"

if [[ -z "$SEARCH_TERM" ]]; then QUERY="$BASE_SQL ORDER BY files.mtime DESC;" RESULTS=$(sqlite3 -separator $'\t' "$DB_PATH" "$QUERY" | tr -d '"') else SAFE_TERM="${SEARCH_TERM//'/''}" # Search both primary title AND alias columns QUERY="$BASE_SQL HAVING nodes.title LIKE '%$SAFE_TERM%' OR alias_list LIKE '%$SAFE_TERM%' ORDER BY files.mtime DESC;" RESULTS=$(sqlite3 -separator $'\t' "$DB_PATH" "$QUERY" | tr -d '"') fi

MATCH_COUNT=$(echo "$RESULTS" | sed '/^$/d' | wc -l)

if [[ $MATCH_COUNT -eq 0 ]]; then echo "No Org-roam nodes or aliases matching: "$SEARCH_TERM"" >&2 exit 1 elif [[ $MATCH_COUNT -eq 1 ]]; then SELECTION="$RESULTS" else # Format fzf display: Title + Aliases in col 1/2, live file preview from col 3 SELECTION=$(echo "$RESULTS"
| fzf --exact
--delimiter=$'\t'
--with-nth=1,2
--tiebreak=index
${SEARCH_TERM:+--query="$SEARCH_TERM"}
--preview 'head -n 30 {3}'
--preview-window=right:50%:wrap) fi

if [[ -n "$SELECTION" ]]; then NODE_TITLE=$(echo "$SELECTION" | cut -f1)

# Hand off execution to the Python terminal browser using the selected title
python3 "$PYTHON_NAV_SCRIPT" "$NODE_TITLE"

fi
